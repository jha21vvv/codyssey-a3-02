# [1차]: 파이썬 타입 힌트를 문자열처럼 지연 평가하여 순환 참조와 신형 타입 표기를 안전하게 지원합니다.

# [1차]: 보일러플레이트 없이 설정 클래스를 선언하기 위해 dataclass 데코레이터를 임포트합니다.
# [2차]: 매번 생성자(__init__)를 손으로 쓰지 않게 해주는 자동 양식 복사기를 꺼냅니다.

# [1차]: frozen=True로 불변(immutable) 설정 데이터클래스 AppConfig를 정의합니다.

    learning_rate: float = 0.005

    # [2차]: 최근 며칠(500프레임) 치 기억을 바탕으로 평균 배경을 계산할지 정합니다.
    history: int = 500
    # [1차]: MOG2에서 픽셀이 배경과 얼마나 달라야 전경(움직임)으로 볼지 정하는 마할라노비스 거리 임계값입니다.
    # [2차]: "이 정도 색 차이는 그냥 흔들림, 이 이상은 진짜 움직임!"을 가르는 민감도 눈금입니다.
    var_threshold: float = 16.0
    # [1차]: 모폴로지 열림/닫힘 연산에 사용할 구조 요소 커널의 크기(홀수)입니다.
    # [2차]: 노이즈 먼지를 털어내고 구멍을 메우는 빗자루의 굵기(5x5 픽셀)입니다.
    morph_kernel_size: int = 5
    # [1차]: 노이즈로 무시하지 않고 객체로 인정할 최소 윤곽선 면적(픽셀²)입니다.
    # [2차]: 먼지나 날파리처럼 너무 작은 점은 무시하는 체(거름망)의 최소 구멍 크기입니다.
    min_area: int = 500
    # [1차]: 조명 급변 등으로 화면 전체가 잡히는 오검출을 막기 위한 최대 윤곽선 면적입니다.
    # [2차]: 화면 절반을 덮는 거대한 빛 번짐은 사람으로 착각하지 않도록 거르는 상한선입니다.
    max_area: int = 100000
    # [1차]: 객체가 화면에서 사라진 뒤 ID를 즉시 지우지 않고 기다려줄 최대 프레임 수입니다.
    # [2차]: 손님이 기둥 뒤에 잠깐 숨었을 때 몇 초(15프레임)까지 번호표를 유지해줄지 정합니다.
    max_disappeared: int = 15
    # [1차]: 이전 프레임 객체와 현재 감지 객체를 같은 ID로 묶을 최대 유클리드 거리(픽셀)입니다.
    # [2차]: 한 걸음(1프레임) 사이에 사람이 순간이동할 수 있는 최대 보폭(80픽셀) 제한입니다.
    max_distance: float = 80.0
    # [1차]: 화면에 선으로 그릴 이동 궤적 좌표의 최대 저장 개수입니다.
    # [2차]: 발자국을 무한정 남기면 지저분하므로 최근 50걸음의 꼬리만 기억합니다.
    max_trajectory_len: int = 50
    # [1차]: 직전 속도 벡터로 다음 위치를 예측해 매칭할지 여부(규칙 기반 개선 비교용)입니다.
    # [2차]: "방금 오른쪽으로 뛰었으니 다음에도 오른쪽에 있겠지"라는 관성 예측 규칙 스위치입니다.
    use_velocity_prediction: bool = False
    # [1차]: 특징점 매칭으로 재인식할 기준 물체 이미지 경로(None이면 비활성화)입니다.
    # [2차]: 영상 속에서 찾아낼 수배 물건의 정면 사진 파일 경로입니다.
    target_image: Optional[str] = "data/targets/box.png"
    # [1차]: 특징점 추출 알고리즘 선택('orb' 또는 'sift')입니다.
    # [2차]: 물체의 지문을 따는 방식(빠른 ORB vs 정밀한 SIFT)을 선택합니다.
    feature_method: str = "orb"
    # [1차]: Lowe's Ratio Test에서 1등 매칭이 2등보다 얼마나 뚜렷해야 통과시킬지 정하는 비율입니다.
    # [2차]: 닮은꼴 후보 중 1등이 2등보다 확실히(75% 미만 거리로) 압도적일 때만 합격시킵니다.
    ratio_thresh: float = 0.75
    # [1차]: TARGET DETECTED 배너를 띄우기 위해 필요한 최소 유효 매칭 개수입니다.
    # [2차]: 지문이 최소 15군데 이상 일치해야 "찾았다!"라고 확신하는 기준선입니다.
    min_match_count: int = 15
    # [1차]: 키보드 's' 입력 시 스냅샷 PNG를 저장할 디렉터리 경로입니다.
    # [2차]: 순간 포착 사진을 모아둘 앨범 폴더 이름입니다.
    snapshot_dir: str = "output/snapshots"
    # [1차]: GUI 창(cv2.imshow) 없이 백그라운드/자동 검증 모드로 실행할지 여부입니다.
    # [2차]: 모니터 화면을 띄우지 않고 조용히 결과만 계산하는 무인 실행 스위치입니다.
    headless: bool = False
    # [1차]: 최대 처리 프레임 수 제한(None이면 영상 끝까지 실행)입니다.
    # [2차]: 긴 영화 전체 대신 앞부분 몇 장면만 빠르게 테스트할 때 쓰는 정지 타이머입니다.
    max_frames: Optional[int] = None
    # [1차]: 렌더링된 결과 영상을 MP4 파일로 저장할 경로(None이면 미저장)입니다.
    # [2차]: 분석 결과가 그려진 화면을 새 비디오 파일로 녹화할지 정하는 출력 경로입니다.
    save_video: Optional[str] = None

    # [1차]: dataclass 초기화 직후 자동 호출되어 모든 설정값의 도메인 유효성을 검증합니다.
    # [2차]: 주방에 들어가기 전에 재료가 상하지 않았는지 입구에서 검수하는 체크리스트입니다.
    def __post_init__(self) -> None:
        # [1차]: source가 빈 문자열이거나 공백만 있는지 검사합니다.
        # [2차]: 재생할 영상 주소가 비어 있으면 출발 전에 바로 경고를 울립니다.
        if not self.source or not str(self.source).strip():
            raise ValueError("source must be a non-empty webcam index or video path.")
        # [1차]: 지원하는 배경 차분 방식('mog2', 'absdiff')인지 확인합니다.
        # [2차]: 메뉴판에 없는 배경 지우개 이름을 적었는지 확인합니다.
        if self.bg_method not in {"mog2", "absdiff"}:
            raise ValueError(f"Unsupported bg_method: {self.bg_method}. Use 'mog2' or 'absdiff'.")
        # [1차]: 학습률이 -1.0(자동) 또는 0.0~1.0 범위 안에 있는지 검증합니다.
        # [2차]: 속도 다이얼이 0%~100% 밖의 엉뚱한 숫자로 돌아가 있지 않은지 체크합니다.
        if not (self.learning_rate == -1.0 or 0.0 <= self.learning_rate <= 1.0):
            raise ValueError("learning_rate must be -1.0 (auto) or in [0.0, 1.0].")
        # [1차]: history 프레임 수가 양수인지 검사합니다.
        # [2차]: 기억 길이가 0 이하일 수 없으므로 양수인지 확인합니다.
        if self.history <= 0:
            raise ValueError("history must be positive.")
        # [1차]: var_threshold가 양수인지 검사합니다.
        # [2차]: 민감도 눈금이 0 이하로 내려가지 않았는지 확인합니다.
        if self.var_threshold <= 0:
            raise ValueError("var_threshold must be positive.")
        # [1차]: 모폴로지 커널 크기가 양의 홀수(3, 5, 7 등)인지 검증합니다.
        # [2차]: 중심 픽셀이 딱 가운데에 오려면 빗자루 크기가 반드시 홀수여야 합니다.
        if self.morph_kernel_size <= 0 or self.morph_kernel_size % 2 == 0:
            raise ValueError("morph_kernel_size must be a positive odd integer.")
        # [1차]: 최소 면적이 양수이며 최대 면적보다 작거나 같은지 검사합니다.
        # [2차]: 체의 최소 구멍이 최대 구멍보다 크면 아무것도 통과 못 하므로 순서를 확인합니다.
        if self.min_area <= 0 or self.max_area < self.min_area:
            raise ValueError("min_area must be positive and <= max_area.")
        # [1차]: max_disappeared가 음수가 아닌지 검증합니다.
        # [2차]: 기다려주는 프레임 수가 마이너스일 수는 없으므로 0 이상인지 확인합니다.
        if self.max_disappeared < 0:
            raise ValueError("max_disappeared must be non-negative.")
        # [1차]: max_distance가 양수인지 검증합니다.
        # [2차]: 최대 이동 보폭은 0보다 커야 매칭이 가능합니다.
        if self.max_distance <= 0:
            raise ValueError("max_distance must be positive.")
        # [1차]: max_trajectory_len이 양수인지 검증합니다.
        # [2차]: 궤적은 최소 1개 이상의 점을 담을 수 있어야 합니다.
        if self.max_trajectory_len <= 0:
            raise ValueError("max_trajectory_len must be positive.")
        # [1차]: feature_method가 'orb' 또는 'sift'인지 검증합니다.
        # [2차]: 지원하는 지문 감식 방식인지 확인합니다.
        if self.feature_method not in {"orb", "sift"}:
            raise ValueError(f"Unsupported feature_method: {self.feature_method}. Use 'orb' or 'sift'.")
        # [1차]: ratio_thresh가 0.0 초과 1.0 미만인지 검증합니다.
        # [2차]: 1등과 2등 비교 비율은 0과 1 사이의 소수여야 의미가 있습니다.
        if not (0.0 < self.ratio_thresh < 1.0):
            raise ValueError("ratio_thresh must be strictly between 0.0 and 1.0.")
        # [1차]: min_match_count가 양수인지 검증합니다.
        # [2차]: 최소 일치 지문 수는 1개 이상이어야 합니다.
        if self.min_match_count <= 0:
            raise ValueError("min_match_count must be positive.")
        # [1차]: max_frames가 지정된 경우 양수인지 검증합니다.
        # [2차]: 프레임 제한을 걸었다면 최소 1프레임 이상 처리하도록 확인합니다.
        if self.max_frames is not None and self.max_frames <= 0:
            raise ValueError("max_frames must be positive when specified.")

"""Background subtraction and morphological noise removal."""

# [1차]: 파이썬 타입 힌트의 지연 평가를 활성화합니다.
# [2차]: 타입 이름표를 미리 평가하지 않고 실행 시 유연하게 처리합니다.
from __future__ import annotations

# [1차]: None 허용 타입 힌트 Optional을 가져옵니다.
# [2차]: 내부 배경 모델이 초기에는 비어 있을 수 있음을 표시합니다.
from typing import Optional

# [1차]: 고전적 영상 처리(MOG2, absdiff, 모폴로지 연산)를 위해 OpenCV를 임포트합니다.
# [2차]: 사진·영상 편집 공구함(OpenCV)을 테이블 위에 펼쳐놓습니다.
import cv2

# [1차]: 영상 프레임(다차원 배열) 연산을 위해 NumPy를 임포트합니다.
# [2차]: 픽셀 숫자 행렬을 초고속으로 계산하는 전자계산기(NumPy)를 꺼냅니다.
import numpy as np


# [1차]: 정지 배경과 움직이는 전경을 분리하는 BackgroundSubtractor 클래스를 정의합니다.
# [2차]: CCTV 화면에서 가만히 있는 배경은 까맣게 지우고 움직이는 사람만 하얗게 남기는 필터 기계입니다.
class BackgroundSubtractor:
    """Separates moving foreground objects from a static or slowly changing background."""

    # [1차]: 배경 차분 방식, 학습률, MOG2 파라미터, 모폴로지 커널 크기를 초기화합니다.
    # [2차]: 필터 기계를 조립하며 지우개 종류, 배경 적응 속도, 빗자루 크기를 세팅합니다.
    def __init__(
        self,
        method: str = "mog2",
        learning_rate: float = 0.005,
        history: int = 500,
        var_threshold: float = 16.0,
        morph_kernel_size: int = 5,
        diff_threshold: int = 25,
    ) -> None:
        # [1차]: 지원하지 않는 배경 차분 방식이 들어오면 ValueError를 발생시킵니다.
        # [2차]: MOG2나 absdiff가 아닌 엉뚱한 버튼을 누르면 즉시 작동을 멈추고 알립니다.
        if method not in {"mog2", "absdiff"}:
            raise ValueError(f"Unsupported background subtraction method: {method}")
        # [1차]: learning_rate가 -1.0 또는 [0.0, 1.0] 범위인지 검증합니다.
        # [2차]: 배경 학습 속도가 허용 범위를 벗어나면 오류를 냅니다.
        if not (learning_rate == -1.0 or 0.0 <= learning_rate <= 1.0):
            raise ValueError("learning_rate must be -1.0 or within [0.0, 1.0].")
        # [1차]: 모폴로지 커널 크기가 양의 홀수인지 확인합니다.
        # [2차]: 대칭이 맞는 홀수 크기 빗자루만 받도록 검사합니다.
        if morph_kernel_size <= 0 or morph_kernel_size % 2 == 0:
            raise ValueError("morph_kernel_size must be a positive odd integer.")

        # [1차]: 선택된 알고리즘, 학습률, 차영상 임계값을 인스턴스 속성에 저장합니다.
        # [2차]: 기계 본체에 설정값을 메모해 둡니다.
        self.method = method
        self.learning_rate = learning_rate
        self.diff_threshold = diff_threshold
        # [1차]: 타원형(MORPH_ELLIPSE) 모폴로지 구조 요소 커널을 생성합니다.
        # [2차]: 둥글둥글한 사람 실루엣을 자연스럽게 다듬기 위해 둥근 모양 스탬프를 만듭니다.
        self.kernel = cv2.getStructuringElement(
            cv2.MORPH_ELLIPSE, (morph_kernel_size, morph_kernel_size)
        )
        # [1차]: method가 'mog2'일 때 OpenCV MOG2 가우시안 혼합 배경 모델 객체를 생성합니다.
        # [2차]: 각 픽셀의 평소 색깔 분포를 통계적으로 외우는 MOG2 두뇌를 장착합니다.
        self._mog2 = (
            cv2.createBackgroundSubtractorMOG2(
                history=history,
                varThreshold=var_threshold,
                detectShadows=True,
            )
            if method == "mog2"
            else None
        )
        # [1차]: absdiff 모드에서 이동 평균 배경을 저장할 float32 배열을 None으로 초기화합니다.
        # [2차]: absdiff 방식이 첫 프레임을 기준 배경 사진으로 담아둘 빈 액자입니다.
        self._bg_model: Optional[np.ndarray] = None

    # [1차]: 입력 BGR/그레이스케일 프레임에서 노이즈가 제거된 이진 전경 마스크(0 또는 255)를 반환합니다.
    # [2차]: 새 CCTV 장면이 들어오면 움직이는 물체만 하얗게 칠한 흑백 도장을 찍어줍니다.
    def apply(self, frame: np.ndarray) -> np.ndarray:
        """Compute a clean binary foreground mask (0 or 255) for the given BGR/gray frame."""
        # [1차]: 입력 프레임이 None이거나 NumPy 배열이 아니거나 비어 있으면 ValueError를 발생시킵니다.
        # [2차]: 카메라에서 빈 종이가 들어오면 고장 나기 전에 입구에서 차단합니다.
        if frame is None or not isinstance(frame, np.ndarray) or frame.size == 0:
            raise ValueError("Input frame must be a non-empty numpy array.")

        # [1차]: MOG2 모드일 때 learningRate를 전달해 전경 마스크를 계산합니다.
        # [2차]: MOG2 두뇌에게 새 화면을 보여주고 "평소 배경과 다른 곳만 표시해!"라고 시킵니다.
        if self.method == "mog2":
            raw_mask = self._mog2.apply(frame, learningRate=self.learning_rate)
            # [1차]: MOG2가 그림자를 회색(127)으로 표시하므로 200 기준 이진화로 그림자를 제거합니다.
            # [2차]: 사람 발밑의 흐릿한 그림자(127)는 지우고 진짜 몸통(255)만 남깁니다.
            _, binary_mask = cv2.threshold(raw_mask, 200, 255, cv2.THRESH_BINARY)
        else:
            # [1차]: 3채널 컬러 영상이면 그레이스케일로 변환하고, 이미 2D면 복사합니다.
            # [2차]: 밝기 차이만 빠르게 비교하기 위해 컬러 사진을 흑백 사진으로 바꿉니다.
            gray = (
                cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
                if frame.ndim == 3
                else frame.copy()
            )
            # [1차]: 5x5 가우시안 블러로 미세한 센서 노이즈를 부드럽게 완화합니다.
            # [2차]: 카메라의 자글자글한 모래알 잡음을 살짝 흐리게 문질러 없앱니다.
            blurred = cv2.GaussianBlur(gray, (5, 5), 0)
            # [1차]: 첫 프레임은 기준 배경 모델로 등록하고 빈 마스크를 반환합니다.
            # [2차]: 비교할 기준 사진이 아직 없으니 첫 장면을 기준 배경으로 액자에 끼워둡니다.
            if self._bg_model is None:
                self._bg_model = blurred.astype(np.float32)
                return np.zeros_like(blurred, dtype=np.uint8)

            # [1차]: 누적 배경 모델(float32)을 uint8로 변환해 현재 프레임과 절대 차이(absdiff)를 구합니다.
            # [2차]: 기준 배경 사진과 지금 사진을 겹쳐 놓고 달라진 픽셀만 뺄셈으로 찾아냅니다.
            bg_uint8 = cv2.convertScaleAbs(self._bg_model)
            diff = cv2.absdiff(bg_uint8, blurred)
            # [1차]: 차이값이 diff_threshold를 넘는 픽셀만 255(전경)로 이진화합니다.
            # [2차]: 미세한 차이는 무시하고 확실히 달라진 곳만 흰색 형광펜으로 칠합니다.
            _, binary_mask = cv2.threshold(
                diff, self.diff_threshold, 255, cv2.THRESH_BINARY
            )
            # [1차]: 학습률(alpha)에 따라 지수이동평균(accumulateWeighted)으로 배경 모델을 서서히 갱신합니다.
            # [2차]: 시간이 지날수록 현재 장면을 아주 조금씩 기준 배경에 섞어 최신화합니다.
            alpha = 0.01 if self.learning_rate < 0 else self.learning_rate
            if alpha > 0.0:
                cv2.accumulateWeighted(blurred, self._bg_model, alpha)

        # [1차]: 모폴로지 열림(MORPH_OPEN: 침식 후 팽창)으로 흩어진 점 노이즈를 제거합니다.
        # [2차]: 빗자루로 바닥에 떨어진 작은 흰 먼지들을 먼저 싹 쓸어냅니다.
        opened = cv2.morphologyEx(binary_mask, cv2.MORPH_OPEN, self.kernel)
        # [1차]: 모폴로지 닫힘(MORPH_CLOSE: 팽창 후 침식)으로 객체 내부의 빈 구멍을 메웁니다.
        # [2차]: 사람 옷 색깔 때문에 몸통 안쪽에 뚫린 작은 구멍들을 퍼티로 매끈하게 채워 하나의 덩어리로 만듭니다.
        cleaned = cv2.morphologyEx(opened, cv2.MORPH_CLOSE, self.kernel)
        # [1차]: 정제된 최종 이진 마스크를 반환합니다.
        # [2차]: 깔끔하게 손질된 흑백 실루엣을 다음 단계(객체 감지기)로 넘겨줍니다.
        return cleaned

    # [1차]: 내부 배경 모델을 초기 상태로 리셋합니다.
    # [2차]: 지금까지 외운 배경 기억을 싹 지우고 새 출발합니다.
    def reset(self) -> None:
        """Reset internal background model state."""
        # [1차]: MOG2 모드라면 기존 하이퍼파라미터를 유지한 채 새 MOG2 인스턴스로 교체합니다.
        # [2차]: 설정은 그대로 두고 MOG2의 기억 메모리만 새것으로 갈아 끼웁니다.
        if self.method == "mog2" and self._mog2 is not None:
            self._mog2 = cv2.createBackgroundSubtractorMOG2(
                history=self._mog2.getHistory(),
                varThreshold=self._mog2.getVarThreshold(),
                detectShadows=True,
            )
        # [1차]: absdiff용 배경 모델 참조를 None으로 비웁니다.
        # [2차]: absdiff 액자 속 기준 사진도 비워둡니다.
        self._bg_model = None

"""Contour-based motion object detection with area filtering."""

# [1차]: 타입 힌트 지연 평가를 활성화합니다.
# [2차]: 타입 표기를 깔끔하고 안전하게 처리합니다.
from __future__ import annotations

# [1차]: 감지 결과를 담을 불변 구조체 생성을 위해 dataclass를 임포트합니다.
# [2차]: 발견한 물체의 좌표와 넓이를 적어둘 결과 카드 양식을 가져옵니다.
from dataclasses import dataclass

# [1차]: 리스트와 튜플 타입 힌트를 임포트합니다.
# [2차]: "목록(List)과 묶음(Tuple) 형태로 돌려준다"는 약속을 적기 위한 도구입니다.
from typing import List, Tuple

# [1차]: 윤곽선 추출(findContours), 면적 계산(contourArea), 바운딩 박스(boundingRect)를 위해 OpenCV를 임포트합니다.
# [2차]: 흰 덩어리의 테두리를 따라 그리고 크기를 재는 도구를 꺼냅니다.
import cv2

# [1차]: 입력 이진 마스크 타입 검증을 위해 NumPy를 임포트합니다.
# [2차]: 마스크가 올바른 픽셀 배열인지 확인하는 검사기입니다.
import numpy as np


# [1차]: 감지된 단일 움직임 영역의 바운딩 박스, 중심점, 면적을 저장하는 불변 데이터클래스입니다.
# [2차]: 찾은 물체 1개마다 "사각형 위치, 한가운데 점, 덩치 크기"를 적어두는 신분증 카드입니다.
@dataclass(frozen=True)
class Detection:
    """Single detected foreground region."""

    # [1차]: 객체의 좌상단 좌표와 너비·높이 (x, y, w, h) 튜플입니다.
    # [2차]: 물체를 감싸는 네모 상자의 위치와 크기입니다.
    bbox: Tuple[int, int, int, int]
    # [1차]: 바운딩 박스의 기하학적 중심점 (cx, cy) 정수 좌표입니다.
    # [2차]: 추적기가 거리를 잴 때 사용할 물체의 배꼽(중심점) 좌표입니다.
    centroid: Tuple[int, int]
    # [1차]: 윤곽선 내부의 실제 픽셀 면적(float)입니다.
    # [2차]: 이 물체가 얼마나 큰 덩어리인지 나타내는 넓이 숫자입니다.
    area: float


# [1차]: 이진 전경 마스크에서 윤곽선을 추출하고 면적 필터링을 수행하는 MotionDetector 클래스입니다.
# [2차]: 흑백 실루엣 그림에서 진짜 움직이는 물체만 골라 네모 박스를 쳐주는 관찰자입니다.
class MotionDetector:
    """Extracts moving object bounding boxes and centroids from a binary foreground mask."""

    # [1차]: 면적 필터링의 하한(min_area)과 상한(max_area)을 검증 후 저장합니다.
    # [2차]: 너무 작은 먼지와 너무 거대한 화면 전체 번짐을 거를 체의 크기를 설정합니다.
    def __init__(self, min_area: int = 500, max_area: int = 100000) -> None:
        # [1차]: 면적 범위가 논리적으로 유효하지 않으면 ValueError를 발생시킵니다.
        # [2차]: 최소 크기가 0 이하이거나 최대 크기보다 크면 잘못된 체이므로 거부합니다.
        if min_area <= 0 or max_area < min_area:
            raise ValueError("min_area must be positive and <= max_area.")
        self.min_area = min_area
        self.max_area = max_area

    # [1차]: 2D 이진 마스크에서 외곽선을 찾고 면적 필터링 후 결정론적으로 정렬된 Detection 목록을 반환합니다.
    # [2차]: 하얀 덩어리들의 테두리를 따서 크기가 적당한 것만 추린 뒤 왼쪽-위 순서로 줄 세워 보고합니다.
    def detect(self, fg_mask: np.ndarray) -> List[Detection]:
        """Find external contours, filter by area, and return deterministic sorted detections."""
        # [1차]: 입력 마스크가 비어 있거나 배열이 아니면 ValueError를 발생시킵니다.
        # [2차]: 빈 도화지가 들어오면 작업을 멈추고 알립니다.
        if fg_mask is None or not isinstance(fg_mask, np.ndarray) or fg_mask.size == 0:
            raise ValueError("fg_mask must be a non-empty 2D numpy array.")
        # [1차]: 3채널 컬러 영상이 잘못 들어오는 것을 막기 위해 2차원(단일 채널)인지 검사합니다.
        # [2차]: 흑백(2D) 마스크가 아니라 컬러(3D) 사진이 실수로 들어오면 거절합니다.
        if fg_mask.ndim != 2:
            raise ValueError("fg_mask must be a single-channel 2D binary image.")

        # [1차]: RETR_EXTERNAL로 최외곽 윤곽선만 추출하고 CHAIN_APPROX_SIMPLE로 꼭짓점을 압축 저장합니다.
        # [2차]: 덩어리 안쪽의 자잘한 구멍은 무시하고 바깥 테두리 선만 깔끔하게 따라 그립니다.
        contours, _ = cv2.findContours(
            fg_mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE
        )

        # [1차]: 면적 조건을 통과한 감지 결과를 담을 리스트를 초기화합니다.
        # [2차]: 합격한 물체 신분증을 모아둘 빈 바구니를 준비합니다.
        detections: List[Detection] = []
        # [1차]: 추출된 모든 윤곽선을 순회하며 면적을 계산하고 필터링합니다.
        # [2차]: 찾은 흰 덩어리를 하나씩 저울에 올려 적당한 크기인지 확인합니다.
        for contour in contours:
            # [1차]: 윤곽선 내부 픽셀 면적을 부동소수점으로 계산합니다.
            # [2차]: 이 덩어리가 몇 픽셀짜리인지 넓이를 잽니다.
            area = float(cv2.contourArea(contour))
            # [1차]: 최소 면적 미만(노이즈) 또는 최대 면적 초과(조명 붕괴 등)는 건너뜁니다.
            # [2차]: 너무 작으면 먼지, 너무 크면 조명 번짐이므로 과감히 탈락시킵니다.
            if area < self.min_area or area > self.max_area:
                continue

            # [1차]: 윤곽선을 감싸는 최소 직교 사각형(x, y, w, h)을 계산합니다.
            # [2차]: 물체에 딱 맞는 직사각형 택배 상자 좌표를 구합니다.
            x, y, w, h = cv2.boundingRect(contour)
            # [1차]: 바운딩 박스의 가로·세로 중앙 좌표(cx, cy)를 정수로 산출합니다.
            # [2차]: 택배 상자의 정중앙에 빨간 핀을 꽂아 중심점을 구합니다.
            cx = int(x + w / 2)
            cy = int(y + h / 2)
            # [1차]: Detection 인스턴스를 생성해 결과 리스트에 추가합니다.
            # [2차]: 합격한 물체의 신분증을 작성해 바구니에 담습니다.
            detections.append(
                Detection(
                    bbox=(int(x), int(y), int(w), int(h)),
                    centroid=(cx, cy),
                    area=area,
                )
            )

        # [1차]: (중심 X, 중심 Y, -면적) 기준으로 정렬하여 매 실행마다 동일한 순서를 보장합니다.
        # [2차]: 매번 실행할 때마다 순서가 뒤죽박죽되지 않도록 왼쪽→오른쪽, 큰 덩어리 순으로 줄을 세웁니다.
        detections.sort(key=lambda d: (d.centroid[0], d.centroid[1], -d.area))
        # [1차]: 정렬된 감지 객체 리스트를 반환합니다.
        # [2차]: 정리된 물체 신분증 바구니를 추적기(Tracker)에게 전달합니다.
        return detections

"""Distance-based Multi-Object Tracker separated into its own module (tracker.py)."""

# [1차]: 타입 힌트의 지연 평가를 활성화합니다.
# [2차]: 클래스와 타입 이름을 유연하게 참조할 수 있게 합니다.
from __future__ import annotations

# [1차]: 최대 길이가 자동으로 제한되는 양방향 큐 deque를 임포트하여 궤적 메모리를 관리합니다.
# [2차]: 새 발자국이 찍히면 가장 오래된 발자국을 자동으로 밀어내는 컨베이어 벨트입니다.
from collections import deque

# [1차]: 추적 객체 상태를 표현하기 위해 dataclass와 가변 기본값 팩토리 field를 임포트합니다.
# [2차]: 객체마다 독립된 궤적 노트(deque)를 안전하게 나눠주기 위한 양식 도구입니다.
from dataclasses import dataclass, field

# [1차]: 두 점 사이의 유클리드 거리 계산(math.hypot)을 위해 math 표준 모듈을 임포트합니다.
# [2차]: 피타고라스 정리로 두 좌표 사이의 직선 거리를 재는 줄자입니다.
import math

# [1차]: 컬렉션 및 시퀀스 타입 힌트를 임포트합니다.
# [2차]: 어떤 상자에 어떤 자료가 들어가는지 표시하는 안내 라벨입니다.
from typing import Deque, Dict, List, Sequence, Tuple

# [1차]: 감지기 모듈에서 전달받는 Detection 데이터클래스를 임포트합니다.
# [2차]: 앞 단계 관찰자가 넘겨주는 물체 신분증 양식을 가져옵니다.
from src.detector import Detection


# [1차]: 추적 중인 개별 객체의 ID, 위치, 박스, 미등장 횟수, 궤적, 속도를 담는 데이터클래스입니다.
# [2차]: 번호표를 받은 손님 1명의 현재 위치, 안 보인 시간, 지나온 발자국, 뛰는 속도를 적은 수첩입니다.
@dataclass
class TrackedObject:
    """Represents a single tracked object with ID, bounding box, trajectory, and velocity."""

    # [1차]: 객체에 부여된 고유 정수 식별자입니다.
    # [2차]: 손님 가슴에 붙여둔 고유 번호표(ID 1, ID 2...)입니다.
    object_id: int
    # [1차]: 현재 프레임에서의 중심점 (cx, cy)입니다.
    # [2차]: 지금 이 손님이 서 있는 위치 좌표입니다.
    centroid: Tuple[int, int]
    # [1차]: 현재 프레임에서의 바운딩 박스 (x, y, w, h)입니다.
    # [2차]: 화면에 그릴 손님 테두리 사각형입니다.
    bbox: Tuple[int, int, int, int]
    # [1차]: 연속으로 화면에서 감지되지 않은 프레임 누적 횟수입니다.
    # [2차]: 손님이 시야에서 사라진 뒤 몇 프레임째 안 보이는지 세는 타이머입니다.
    disappeared: int = 0
    # [1차]: 최근 중심점 이동 경로를 저장하는 고정 길이 deque입니다.
    # [2차]: 손님이 걸어온 발자국 좌표를 최대 50개까지 모아두는 꼬리입니다.
    trajectory: Deque[Tuple[int, int]] = field(default_factory=lambda: deque(maxlen=50))
    # [1차]: 직전 프레임 대비 중심점 이동 벡터 (vx, vy)입니다.
    # [2차]: 방금 1프레임 사이에 가로·세로로 몇 픽셀 움직였는지 적어둔 속도계입니다.
    velocity: Tuple[float, float] = (0.0, 0.0)


# [1차]: 유클리드 거리 기반 최근접 매칭으로 다중 객체를 추적하는 독립 Tracker 클래스입니다.
# [2차]: 이전 화면의 손님들과 지금 화면의 손님들 사이 거리를 재어 가장 가까운 사람끼리 같은 번호표를 이어주는 안내원입니다.
class Tracker:
    """Euclidean distance-based centroid tracker with trajectory storage and optional velocity prediction."""

    # [1차]: 미등장 허용 프레임, 최대 매칭 거리, 궤적 길이, 속도 예측 사용 여부를 검증 후 초기화합니다.
    # [2차]: 안내원의 규칙(몇 초 기다릴지, 최대 보폭, 꼬리 길이, 관성 예측 여부)을 설정하고 출석부를 펼칩니다.
    def __init__(
        self,
        max_disappeared: int = 15,
        max_distance: float = 80.0,
        max_trajectory_len: int = 50,
        use_velocity_prediction: bool = False,
    ) -> None:
        # [1차]: 잘못된 하이퍼파라미터 입력을 즉시 차단합니다.
        # [2차]: 말이 안 되는 숫자(음수 대기 시간, 0 이하 보폭 등)가 들어오면 바로 경고를 냅니다.
        if max_disappeared < 0:
            raise ValueError("max_disappeared must be non-negative.")
        if max_distance <= 0:
            raise ValueError("max_distance must be positive.")
        if max_trajectory_len <= 0:
            raise ValueError("max_trajectory_len must be positive.")

        self.max_disappeared = max_disappeared
        self.max_distance = max_distance
        self.max_trajectory_len = max_trajectory_len
        self.use_velocity_prediction = use_velocity_prediction

        # [1차]: 다음에 부여할 고유 ID 카운터(1부터 단조 증가)를 초기화합니다.
        # [2차]: 번호표 기계의 다음 숫자를 1번으로 맞춰둡니다.
        self.next_object_id: int = 1
        # [1차]: 현재 활성 추적 객체를 `{object_id: TrackedObject}` 딕셔너리로 관리합니다.
        # [2차]: 지금 매장에 있는 손님들의 출석부입니다.
        self.objects: Dict[int, TrackedObject] = {}
        # [1차]: 전체 실행 기간 동안 누적 등록된 고유 ID 총 개수를 기록합니다.
        # [2차]: 오늘 하루 동안 총 몇 명의 손님에게 번호표를 뽑아주었는지 세는 카운터입니다.
        self.total_registered: int = 0

    # [1차]: 새로 등장한 감지 객체에 고유 ID를 발급하고 출석부(self.objects)에 등록합니다.
    # [2차]: 새로 들어온 손님에게 새 번호표를 뽑아주고 첫 발자국을 기록합니다.
    def register(self, detection: Detection) -> TrackedObject:
        """Assign a new unique ID to a newly appeared object."""
        # [1차]: 현재 ID를 할당하고 다음 ID와 누적 등록 카운터를 1씩 증가시킵니다.
        # [2차]: 번호표를 한 장 뜯어주고 기계 숫자를 다음 번호로 넘깁니다.
        obj_id = self.next_object_id
        self.next_object_id += 1
        self.total_registered += 1

        # [1차]: 최대 길이 제한이 걸린 궤적 큐를 만들고 시작 중심점을 삽입합니다.
        # [2차]: 새 발자국 수첩을 열어 첫 번째 위치를 적어둡니다.
        traj: Deque[Tuple[int, int]] = deque(maxlen=self.max_trajectory_len)
        traj.append(detection.centroid)
        # [1차]: TrackedObject 인스턴스를 생성해 딕셔너리에 저장하고 반환합니다.
        # [2차]: 손님 정보를 출석부에 정식 등록합니다.
        tracked = TrackedObject(
            object_id=obj_id,
            centroid=detection.centroid,
            bbox=detection.bbox,
            disappeared=0,
            trajectory=traj,
            velocity=(0.0, 0.0),
        )
        self.objects[obj_id] = tracked
        return tracked

    # [1차]: 지정된 object_id를 추적 목록에서 안전하게 제거합니다(없어도 예외 없음).
    # [2차]: 매장을 떠난 손님의 번호표를 출석부에서 지워줍니다.
    def deregister(self, object_id: int) -> None:
        """Remove an object ID that has disappeared beyond the allowed threshold."""
        self.objects.pop(object_id, None)

    # [1차]: 거리 비교에 사용할 기준 좌표를 계산합니다(속도 예측 ON 시 현재 위치 + 직전 속도).
    # [2차]: 기본 모드는 "방금 서 있던 곳", 속도 예측 모드는 "뛰던 방향으로 한 걸음 더 간 예상 위치"를 기준으로 삼습니다.
    def _reference_point(self, obj: TrackedObject) -> Tuple[float, float]:
        """Compute matching reference point (current centroid or constant-velocity prediction)."""
        if self.use_velocity_prediction and len(obj.trajectory) >= 2:
            return (
                obj.centroid[0] + obj.velocity[0],
                obj.centroid[1] + obj.velocity[1],
            )
        return (float(obj.centroid[0]), float(obj.centroid[1]))

    # [1차]: math.hypot을 이용해 두 2D 점 사이의 유클리드 직선 거리를 수치적으로 안정하게 계산합니다.
    # [2차]: 가로 차이와 세로 차이로 대각선 직선 거리를 재는 피타고라스 줄자입니다.
    @staticmethod
    def _euclidean_distance(
        p1: Tuple[float, float], p2: Tuple[float, float]
    ) -> float:
        return math.hypot(p1[0] - p2[0], p1[1] - p2[1])

    # [1차]: 현재 프레임의 감지 목록을 받아 기존 객체와 1:1 최근접 매칭·갱신·삭제·신규등록을 수행합니다.
    # [2차]: 새 장면이 올 때마다 기존 손님과 새 위치들을 짝지어 출석부를 최신화하는 핵심 두뇌입니다.
    def update(self, detections: Sequence[Detection]) -> Dict[int, TrackedObject]:
        """Update tracked objects for the current frame using greedy nearest-distance matching."""
        # [1차]: 현재 프레임에 감지된 객체가 0개면 모든 기존 객체의 미등장 카운터를 1씩 증가시킵니다.
        # [2차]: 화면이 텅 비었으면 기존 손님들의 "안 보인 시간"을 1씩 올리고 너무 오래 안 보이면 퇴장 처리합니다.
        if len(detections) == 0:
            for obj_id in list(self.objects.keys()):
                self.objects[obj_id].disappeared += 1
                if self.objects[obj_id].disappeared > self.max_disappeared:
                    self.deregister(obj_id)
            return dict(self.objects)

        # [1차]: 기존에 추적 중인 객체가 하나도 없으면 이번 감지를 모두 신규 등록합니다.
        # [2차]: 출석부가 비어 있으니 지금 보이는 사람들에게 차례로 새 번호표를 나눠줍니다.
        if len(self.objects) == 0:
            for det in detections:
                self.register(det)
            return dict(self.objects)

        # [1차]: 결정론적 순서를 위해 기존 ID를 오름차순 정렬하고 모든 (기존객체, 신규감지) 쌍의 거리를 계산합니다.
        # [2차]: 모든 기존 손님과 새 위치 조합 사이의 거리를 줄자로 하나씩 재어 표에 적어둡니다.
        object_ids = sorted(self.objects.keys())
        pairs: List[Tuple[float, int, int]] = []
        for obj_id in object_ids:
            ref_pt = self._reference_point(self.objects[obj_id])
            for det_idx, det in enumerate(detections):
                dist = self._euclidean_distance(
                    ref_pt, (float(det.centroid[0]), float(det.centroid[1]))
                )
                # [1차]: (거리, object_id, det_idx) 튜플로 저장하여 동률 거리 시 낮은 ID가 우선 매칭되게 합니다.
                # [2차]: 거리가 똑같이 10px로 비길 때 운에 맡기지 않고 먼저 온 번호표(작은 ID)에게 우선권을 줍니다.
                pairs.append((dist, obj_id, det_idx))

        # [1차]: 거리 오름차순(동률 시 ID, 인덱스 오름차순)으로 후보 쌍을 정렬합니다.
        # [2차]: 가장 가까운 커플 후보부터 차례대로 줄을 세웁니다.
        pairs.sort(key=lambda item: (item[0], item[1], item[2]))

        # [1차]: 1:1 매칭을 보장하기 위해 이미 짝지어진 기존 ID와 신규 감지 인덱스를 집합으로 추적합니다.
        # [2차]: 이미 짝을 찾은 손님과 위치에 "매칭 완료" 도장을 찍어 중복 배정을 막습니다.
        used_object_ids = set()
        used_det_indices = set()

        # [1차]: 가장 가까운 쌍부터 탐욕(Greedy) 방식으로 1:1 매칭을 확정합니다.
        # [2차]: 줄 선 후보 중 제일 가까운 쌍부터 짝을 지어줍니다.
        for dist, obj_id, det_idx in pairs:
            # [1차]: 정렬되어 있으므로 현재 거리가 max_distance를 넘으면 이후 쌍도 모두 초과하므로 즉시 중단합니다.
            # [2차]: 최대 보폭보다 먼 후보는 같은 사람일 수 없으므로 여기서 매칭을 끝냅니다.
            if dist > self.max_distance:
                break
            # [1차]: 이미 매칭된 객체나 감지는 건너뜁니다.
            # [2차]: 둘 중 하나라도 이미 짝이 있으면 패스합니다.
            if obj_id in used_object_ids or det_idx in used_det_indices:
                continue

            # [1차]: 매칭 확정된 객체의 속도, 중심점, 바운딩 박스, 미등장 카운터, 궤적을 갱신합니다.
            # [2차]: "아, 아까 그 손님이 여기로 이동했구나!" 하고 위치·속도·발자국을 업데이트합니다.
            det = detections[det_idx]
            obj = self.objects[obj_id]
            vx = float(det.centroid[0] - obj.centroid[0])
            vy = float(det.centroid[1] - obj.centroid[1])
            obj.velocity = (vx, vy)
            obj.centroid = det.centroid
            obj.bbox = det.bbox
            obj.disappeared = 0
            obj.trajectory.append(det.centroid)

            used_object_ids.add(obj_id)
            used_det_indices.add(det_idx)

        # [1차]: 이번 프레임에서 짝을 찾지 못한 기존 객체는 미등장 카운터를 늘리고 한도 초과 시 삭제합니다.
        # [2차]: 이번에 안 보인 손님은 대기 카운트를 올리고, 너무 오래 안 돌아오면 출석부에서 지웁니다.
        for obj_id in object_ids:
            if obj_id not in used_object_ids:
                self.objects[obj_id].disappeared += 1
                if self.objects[obj_id].disappeared > self.max_disappeared:
                    self.deregister(obj_id)

        # [1차]: 기존 객체와 짝지어지지 않은 신규 감지는 새 ID를 발급해 등록합니다.
        # [2차]: 짝이 남는 새 위치는 새로 입장한 손님이므로 새 번호표를 뽑아줍니다.
        for det_idx, det in enumerate(detections):
            if det_idx not in used_det_indices:
                self.register(det)

        # [1차]: 외부에서 내부 딕셔너리를 직접 변형하지 않도록 얕은 복사본을 반환합니다.
        # [2차]: 원본 출석부는 금고에 두고 복사본을 화가(시각화 모듈)에게 건네줍니다.
        return dict(self.objects)

# [1차]: 단일 프레임에 대한 특징점 매칭 정량 결과를 담는 불변 데이터클래스입니다.
# [2차]: "찾았는지 여부, 기준 지문 수, 화면 지문 수, 일치 개수, 일치율(%)"이 적힌 감식 보고서입니다.
@dataclass(frozen=True)
class MatchResult:
    """Quantitative result of feature matching on a single frame."""

    # [1차]: 유효 매칭 수가 min_match_count 이상인지 여부(TARGET DETECTED 판정)입니다.
    # [2차]: 화면에 수배 물건이 등장했는지(True/False) 도장을 찍는 칸입니다.
    detected: bool
    # [1차]: 등록된 기준 타겟 이미지에서 추출된 특징점 총 개수입니다.
    # [2차]: 수배 전단지 사진에서 뽑아낸 지문 총 개수입니다.
    target_keypoints: int
    # [1차]: 현재 영상 프레임에서 추출된 특징점 총 개수입니다.
    # [2차]: 지금 CCTV 화면 전체에서 뽑아낸 지문 총 개수입니다.
    frame_keypoints: int
    # [1차]: Lowe's Ratio Test를 통과한 신뢰성 있는 매칭 쌍 개수입니다.
    # [2차]: 수배 사진과 CCTV 화면에서 진짜로 똑같다고 판명된 지문 개수입니다.
    good_matches: int
    # [1차]: 기준 특징점 대비 유효 매칭 비율(백분율, 소수 둘째 자리)입니다.
    # [2차]: 수배 지문 중 몇 퍼센트가 화면에서 발견되었는지 보여주는 일치율입니다.
    match_rate: float
    # [1차]: 사용된 특징점 알고리즘 이름('orb' 또는 'sift')입니다.
    # [2차]: 어떤 감식 기법을 썼는지 적어두는 칸입니다.
    method: str


# [1차]: Windows 환경에서 한글·공백 포함 경로의 cv2.imread 실패 버그를 우회하는 안전한 이미지 로더입니다.
# [2차]: OpenCV가 한글 폴더 이름을 못 읽는 약점을 피해 파이썬으로 먼저 바이트를 읽은 뒤 그림으로 복원합니다.
def _imread_unicode(path: Path, flags: int = cv2.IMREAD_GRAYSCALE) -> Optional[np.ndarray]:
    """Read an image safely even when the Windows file path contains Korean or spaces."""
    # [1차]: 파일이 존재하지 않거나 디렉터리라면 None을 반환합니다.
    # [2차]: 사진 파일이 실제로 그 자리에 있는지 먼저 확인합니다.
    if not path.exists() or not path.is_file():
        return None
    # [1차]: 파일의 원시 바이트를 uint8 1차원 NumPy 배열로 읽습니다.
    # [2차]: 파일 이름을 신경 쓰지 않고 사진 파일의 알맹이 숫자 덩어리를 그대로 가져옵니다.
    raw = np.fromfile(str(path), dtype=np.uint8)
    # [1차]: 빈 파일(0바이트)인 경우 None을 반환합니다.
    # [2차]: 속이 텅 빈 파일이면 깨지기 전에 돌려보냅니다.
    if raw.size == 0:
        return None
    # [1차]: 메모리 버퍼의 바이트 배열을 OpenCV 이미지 행렬로 디코딩합니다.
    # [2차]: 숫자 덩어리를 흑백(또는 컬러) 사진으로 펼쳐줍니다.
    return cv2.imdecode(raw, flags)


# [1차]: ORB 또는 SIFT 특징점 매칭으로 특정 물체를 매 프레임 재인식하는 FeatureMatcher 클래스입니다.
# [2차]: 미리 등록해 둔 물건의 지문을 외워두고 CCTV 프레임마다 똑같은 물건이 나타났는지 감시하는 감식반입니다.
class FeatureMatcher:
    """Re-identifies a pre-registered target object image in video frames using ORB or SIFT."""

    # [1차]: 타겟 이미지를 로드하고 선택한 알고리즘(ORB/SIFT)으로 기준 특징점과 디스크립터를 사전 계산합니다.
    # [2차]: 출근하자마자 수배 물건 사진을 받아 지문을 미리 따서 머릿속에 저장해 둡니다.
    def __init__(
        self,
        target_image_path: str | Path,
        method: str = "orb",
        ratio_thresh: float = 0.75,
        min_match_count: int = 15,
        nfeatures: int = 1000,
    ) -> None:
        # [1차]: 알고리즘 이름, 비율 임계값, 최소 매칭 수의 유효성을 검증합니다.
        # [2차]: 감식 방식과 합격 기준이 올바른지 입구에서 확인합니다.
        if method not in {"orb", "sift"}:
            raise ValueError(f"Unsupported feature method: {method}. Use 'orb' or 'sift'.")
        if not (0.0 < ratio_thresh < 1.0):
            raise ValueError("ratio_thresh must be between 0.0 and 1.0.")
        if min_match_count <= 0:
            raise ValueError("min_match_count must be positive.")

        self.target_path = Path(target_image_path)
        self.method = method
        self.ratio_thresh = ratio_thresh
        self.min_match_count = min_match_count

        # [1차]: 유니코드 안전 로더로 기준 이미지를 흑백(그레이스케일)으로 읽어옵니다.
        # [2차]: 수배 물건 사진을 흑백 지문 분석용으로 꺼내옵니다.
        target_gray = _imread_unicode(self.target_path, cv2.IMREAD_GRAYSCALE)
        # [1차]: 이미지 로드 실패 시 FileNotFoundError를 발생시킵니다.
        # [2차]: 수배 사진이 없으면 감식을 시작할 수 없으므로 오류를 알립니다.
        if target_gray is None:
            raise FileNotFoundError(f"Failed to load target image: {self.target_path}")

        # [1차]: ORB는 이진 디스크립터이므로 해밍 거리(NORM_HAMMING), SIFT는 실수 벡터이므로 L2 거리를 사용합니다.
        # [2차]: ORB 지문은 0/1 비트 차이로, SIFT 지문은 직선 거리로 닮은 정도를 재는 전용 자를 준비합니다.
        if method == "orb":
            self._extractor = cv2.ORB_create(nfeatures=nfeatures)
            self._matcher = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=False)
        else:
            if not hasattr(cv2, "SIFT_create"):
                raise RuntimeError("cv2.SIFT_create is not available in this OpenCV build.")
            self._extractor = cv2.SIFT_create(nfeatures=nfeatures)
            self._matcher = cv2.BFMatcher(cv2.NORM_L2, crossCheck=False)

        # [1차]: 기준 이미지에서 특징점(Keypoints)과 기술자(Descriptors)를 미리 추출해 캐싱합니다.
        # [2차]: 매번 수배 사진을 다시 분석하면 느리므로 처음 1번만 지문을 따서 보관해 둡니다.
        self._target_kp, self._target_des = self._extractor.detectAndCompute(
            target_gray, None
        )
        # [1차]: 민무늬 이미지처럼 특징점이 전혀 추출되지 않으면 ValueError를 발생시킵니다.
        # [2차]: 하얀 종이처럼 지문이 하나도 없는 사진은 찾을 수 없으므로 거절합니다.
        if self._target_des is None or len(self._target_kp) == 0:
            raise ValueError(f"No features could be extracted from target image: {self.target_path}")

    # [1차]: 기준 타겟 이미지에서 추출된 특징점 개수를 읽기 전용 속성으로 제공합니다.
    # [2차]: 보관 중인 수배 지문이 총 몇 개인지 알려주는 안내창입니다.
    @property
    def target_keypoint_count(self) -> int:
        return len(self._target_kp)

    # [1차]: 현재 프레임에서 특징점을 추출하고 knnMatch(k=2) + Lowe's Ratio Test로 재인식 결과를 반환합니다.
    # [2차]: CCTV 화면에서 지문을 뽑아 수배 지문과 비교한 뒤 "TARGET DETECTED" 여부를 판정합니다.
    def match(self, frame: np.ndarray) -> MatchResult:
        """Match the registered target image against a BGR or grayscale video frame."""
        # [1차]: 빈 프레임 입력 시 ValueError로 즉시 방어합니다.
        # [2차]: 빈 화면이 들어오면 오류를 알립니다.
        if frame is None or not isinstance(frame, np.ndarray) or frame.size == 0:
            raise ValueError("Input frame must be a non-empty numpy array.")

        # [1차]: 3채널 컬러 영상이면 흑백으로 변환하고, 이미 2D면 그대로 사용합니다.
        # [2차]: 지문 비교를 위해 컬러 화면을 흑백으로 바꿉니다.
        gray = (
            cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            if frame.ndim == 3
            else frame
        )
        # [1차]: 현재 프레임의 특징점과 디스크립터를 추출합니다.
        # [2차]: 지금 CCTV 화면 곳곳에서 지문을 채취합니다.
        frame_kp, frame_des = self._extractor.detectAndCompute(gray, None)

        target_count = len(self._target_kp)
        frame_count = 0 if frame_kp is None else len(frame_kp)

        # [1차]: knnMatch(k=2)는 최소 2개의 비교 대상이 필요하므로 2개 미만이면 안전하게 미검출을 반환합니다.
        # [2차]: 어두운 화면이라 지문이 2개도 안 나오면 억지로 비교하다 뻗지 않고 "못 찾음"으로 안전 보고합니다.
        if frame_des is None or frame_count < 2 or len(self._target_des) < 2:
            return MatchResult(
                detected=False,
                target_keypoints=target_count,
                frame_keypoints=frame_count,
                good_matches=0,
                match_rate=0.0,
                method=self.method,
            )

        # [1차]: 각 타겟 특징점에 대해 가장 가까운 1등(m)과 2등(n) 매칭 후보를 찾습니다.
        # [2차]: 수배 지문마다 CCTV에서 가장 닮은 후보 1등과 2등을 뽑아옵니다.
        knn_matches = self._matcher.knnMatch(self._target_des, frame_des, k=2)
        # [1차]: Lowe's Ratio Test(m.distance < ratio_thresh * n.distance)를 통과한 확실한 매칭만 카운트합니다.
        # [2차]: 1등이 2등과 비슷하게 애매하면 탈락시키고, 1등이 압도적으로 닮았을 때만 진짜 일치로 인정합니다.
        good_count = 0
        for pair in knn_matches:
            if len(pair) == 2:
                m, n = pair
                if m.distance < self.ratio_thresh * n.distance:
                    good_count += 1

        # [1차]: 타겟 특징점 대비 유효 매칭 비율(%)과 임계값 통과 여부를 계산해 반환합니다.
        # [2차]: 일치율(%)을 계산하고 기준 개수(15개) 이상이면 "찾았다(True)" 도장을 찍어 보고합니다.
        match_rate = round((good_count / target_count) * 100.0, 2) if target_count > 0 else 0.0
        detected = good_count >= self.min_match_count

        return MatchResult(
            detected=detected,
            target_keypoints=target_count,
            frame_keypoints=frame_count,
            good_matches=good_count,
            match_rate=match_rate,
            method=self.method,
        )

    # [1차]: 회전(warpAffine) 및 좌측 가림(픽셀 마스킹) 변형 이미지를 생성하는 정량 벤치마크 유틸리티입니다.
    # [2차]: 수배 사진을 30도/60도 돌리거나 손으로 30%/50% 가려보는 모의고사 시험지 제작기입니다.
    @staticmethod
    def transform_image(
        image: np.ndarray,
        angle_deg: float = 0.0,
        occlusion_ratio: float = 0.0,
    ) -> np.ndarray:
        """Create controlled rotation/occlusion variants of an image for quantitative benchmarking."""
        # [1차]: 빈 이미지 입력을 방어하고 원본을 훼손하지 않기 위해 복사본을 만듭니다.
        # [2차]: 원본 사진은 소중하니 복사본을 떠서 실험합니다.
        if image is None or image.size == 0:
            raise ValueError("image must be non-empty.")
        result = image.copy()
        h, w = result.shape[:2]

        # [1차]: 회전 각도가 0이 아니면 이미지 중심을 기준으로 아핀 회전 변환을 적용합니다.
        # [2차]: 사진 한가운데에 압정을 꽂고 지정한 각도만큼 빙그르르 돌려놓습니다.
        if angle_deg != 0.0:
            center = (w / 2.0, h / 2.0)
            rot_mat = cv2.getRotationMatrix2D(center, angle_deg, 1.0)
            result = cv2.warpAffine(
                result,
                rot_mat,
                (w, h),
                flags=cv2.INTER_LINEAR,
                borderMode=cv2.BORDER_CONSTANT,
                borderValue=0,
            )

        # [1차]: 가림 비율(0.0~1.0)만큼 왼쪽 영역의 픽셀을 0(검정)으로 덮어 가림(occlusion)을 재현합니다.
        # [2차]: 사진 왼쪽부터 지정한 비율만큼 검은 종이로 가려 물체가 숨은 상황을 만듭니다.
        if occlusion_ratio > 0.0:
            clamped = min(max(occlusion_ratio, 0.0), 1.0)
            occ_w = int(round(w * clamped))
            result[:, :occ_w] = 0

        return result

# [1차]: 추적 결과, FPS HUD, 타겟 감지 배너를 프레임에 오버레이하고 스냅샷을 저장하는 클래스입니다.
# [2차]: 분석 결과를 사람이 한눈에 볼 수 있게 CCTV 화면 위에 깔끔하게 그려주는 전담 화가입니다.
class FrameVisualizer:
    """Draws bounding boxes, unique IDs, trajectories, FPS HUD, and target detection banners."""

    # [1차]: 객체 ID별로 구분하기 쉬운 8가지 고정 BGR 색상 팔레트를 정의합니다.
    # [2차]: 손님마다 서로 다른 형광펜 색깔을 배정하기 위한 8색 크레파스 통입니다.
    _PALETTE = (
        (0, 220, 0),
        (255, 165, 0),
        (0, 191, 255),
        (255, 105, 180),
        (147, 112, 219),
        (64, 224, 208),
        (255, 215, 0),
        (255, 99, 71),
    )

    # [1차]: object_id에 나머지 연산(%)을 적용해 항상 동일한 BGR 색상을 결정론적으로 반환합니다.
    # [2차]: 1번 손님은 늘 같은 색깔, 2번 손님은 늘 다른 색깔로 일관되게 칠해줍니다.
    @classmethod
    def color_for_id(cls, object_id: int) -> Tuple[int, int, int]:
        """Return a deterministic BGR color for each object ID."""
        return cls._PALETTE[object_id % len(cls._PALETTE)]

    # [1차]: 원본 프레임 복사본 위에 바운딩 박스, ID, 궤적, FPS HUD, 타겟 배너, 일시정지 표시를 렌더링합니다.
    # [2차]: 원본 사진 위에 투명 필름을 얹어 네모 상자·번호·발자국·속도계·경고등을 모두 그려줍니다.
    def draw(
        self,
        frame: np.ndarray,
        tracked_objects: Dict[int, TrackedObject],
        fps: float,
        match_result: Optional[MatchResult] = None,
        paused: bool = False,
        bg_info: str = "",
    ) -> np.ndarray:
        """Return a copy of frame with all tracking overlays and HUD elements drawn."""
        # [1차]: 원본 프레임을 변형하지 않도록 복사본(canvas)을 생성합니다.
        # [2차]: 원본 CCTV 화면을 망치지 않게 새 도화지에 복사해서 그립니다.
        canvas = frame.copy()

        # [1차]: 모든 활성 추적 객체를 순회하며 박스, 중심점, ID 텍스트, 이동 궤적을 그립니다.
        # [2차]: 출석부에 있는 손님 한 명씩 차례로 테두리·번호표·걸어온 길 선을 그려줍니다.
        for obj_id, obj in tracked_objects.items():
            color = self.color_for_id(obj_id)
            x, y, w, h = obj.bbox

            # [1차]: 바운딩 박스 직사각형과 중심점 채워진 원(반지름 4px)을 그립니다.
            # [2차]: 손님 둘레에 네모 박스를 치고 배꼽 위치에 점을 찍습니다.
            cv2.rectangle(canvas, (x, y), (x + w, y + h), color, 2)
            cv2.circle(canvas, obj.centroid, 4, color, -1)

            # [1차]: 바운딩 박스 위쪽(화면 밖으로 잘리지 않도록 최소 y=16 보장)에 ID 라벨을 출력합니다.
            # [2차]: 네모 상자 바로 위에 "ID 1" 이름표를 달아줍니다.
            label = f"ID {obj_id}"
            text_y = max(y - 8, 16)
            cv2.putText(
                canvas,
                label,
                (x, text_y),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.55,
                color,
                2,
                cv2.LINE_AA,
            )

            # [1차]: 궤적 좌표가 2개 이상이면 (N, 1, 2) int32 배열로 변환해 열린 다각형 선(polylines)을 그립니다.
            # [2차]: 발자국이 2개 이상 모이면 점들을 실로 꿰어 지나온 이동 경로를 선으로 보여줍니다.
            if len(obj.trajectory) >= 2:
                pts = np.array(list(obj.trajectory), dtype=np.int32).reshape((-1, 1, 2))
                cv2.polylines(canvas, [pts], isClosed=False, color=color, thickness=2)

        # [1차]: 좌상단 HUD 영역에 반투명 검은 배경 박스(addWeighted alpha=0.65)를 합성합니다.
        # [2차]: 글씨가 잘 보이도록 화면 왼쪽 위에 반투명 검은 유리판을 덧댑니다.
        hud_w = 340
        hud_h = 88 if bg_info else 68
        overlay = canvas.copy()
        cv2.rectangle(overlay, (8, 8), (8 + hud_w, 8 + hud_h), (20, 20, 20), -1)
        cv2.addWeighted(overlay, 0.65, canvas, 0.35, 0, canvas)

        # [1차]: 현재 초당 프레임(FPS)과 추적 중인 객체 수를 HUD 첫 줄에 표시합니다.
        # [2차]: 자동차 계기판처럼 현재 속도와 감지된 사람 수를 띄워줍니다.
        cv2.putText(
            canvas,
            f"FPS: {fps:.1f} | Objects: {len(tracked_objects)}",
            (18, 34),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.62,
            (0, 255, 180),
            2,
            cv2.LINE_AA,
        )
        # [1차]: 키보드 단축키 안내(q 종료, s 스냅샷, p 일시정지)를 둘째 줄에 표시합니다.
        # [2차]: 사용자가 까먹지 않도록 리모컨 버튼 설명을 작게 적어둡니다.
        cv2.putText(
            canvas,
            "Keys: [q] Quit  [s] Snapshot  [p] Pause",
            (18, 58),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.46,
            (220, 220, 220),
            1,
            cv2.LINE_AA,
        )
        # [1차]: 배경 차분 방식 및 학습률 정보가 있으면 셋째 줄에 표시합니다.
        # [2차]: 현재 어떤 배경 지우개와 학습 속도로 돌아가는지 상태 줄에 적어줍니다.
        if bg_info:
            cv2.putText(
                canvas,
                bg_info,
                (18, 80),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.45,
                (180, 220, 255),
                1,
                cv2.LINE_AA,
            )

        # [1차]: 특징점 매칭 성공 시 하단에 초록색 배너와 TARGET DETECTED (Target Found) 메시지를 그립니다.
        # [2차]: 수배 물건이 화면에 나타나면 하단에 커다란 초록색 알림 띠를 띄워 알려줍니다.
        if match_result is not None and match_result.detected:
            h_img, w_img = canvas.shape[:2]
            banner_text = (
                f"TARGET DETECTED (Target Found) | "
                f"{match_result.method.upper()} {match_result.good_matches} ({match_result.match_rate:.1f}%)"
            )
            cv2.rectangle(canvas, (8, h_img - 48), (w_img - 8, h_img - 10), (0, 140, 0), -1)
            cv2.putText(
                canvas,
                banner_text,
                (18, h_img - 22),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.6,
                (255, 255, 255),
                2,
                cv2.LINE_AA,
            )

        # [1차]: 일시정지 상태라면 우상단에 주황색 PAUSED 배지를 표시합니다.
        # [2차]: 리모컨 일시정지 버튼을 누르면 화면 구석에 "일시정지 중" 표시등을 켭니다.
        if paused:
            cv2.putText(
                canvas,
                "PAUSED",
                (canvas.shape[1] - 130, 38),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.85,
                (0, 165, 255),
                2,
                cv2.LINE_AA,
            )

        return canvas

    # [1차]: cv2.imencode + Path.write_bytes 조합으로 한글 경로에서도 깨지지 않게 PNG 스냅샷을 저장합니다.
    # [2차]: OpenCV가 한글 폴더에 저장을 못 하는 문제를 피해 메모리에서 PNG로 압축한 뒤 파이썬이 직접 파일로 써줍니다.
    @staticmethod
    def save_snapshot(frame: np.ndarray, snapshot_dir: str | Path, frame_idx: int) -> Path:
        """Save current rendered frame as PNG safely even with Unicode paths."""
        # [1차]: 저장 폴더가 없으면 부모 경로까지 포함해 자동 생성합니다.
        # [2차]: 사진 앨범 폴더가 없으면 새로 만들어 둡니다.
        out_dir = Path(snapshot_dir)
        out_dir.mkdir(parents=True, exist_ok=True)
        # [1차]: 현재 시각과 프레임 번호를 조합해 중복되지 않는 파일명을 만듭니다.
        # [2차]: "20261125_190500_f00042.png"처럼 고유한 사진 이름을 붙입니다.
        stamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        out_path = out_dir / f"snapshot_{stamp}_f{frame_idx:05d}.png"
        # [1차]: 메모리 버퍼에서 PNG 포맷으로 인코딩하고 실패 시 RuntimeError를 발생시킵니다.
        # [2차]: 픽셀 그림을 PNG 파일 데이터로 압축합니다.
        ok, encoded = cv2.imencode(".png", frame)
        if not ok:
            raise RuntimeError("Failed to encode snapshot frame as PNG.")
        # [1차]: 인코딩된 바이트를 파일로 기록하고 저장 경로를 반환합니다.
        # [2차]: 압축된 사진을 앨범 폴더에 안전하게 저장하고 파일 주소를 돌려줍니다.
        out_path.write_bytes(encoded.tobytes())
        return out_path

"""Real-time video analysis application loop with keyboard controls and headless mode."""

# [1차]: 실행 요약 통계를 담는 dataclass를 임포트합니다.

# [1차]: 타겟 이미지 및 출력 비디오 경로 확인을 위해 Path를 임포트합니다.
# [2차]: 파일이 존재하는지, 저장할 폴더가 있는지 관리하는 도구입니다.

# [1차]: 고정밀 단조 시계(time.perf_counter)로 실시간 FPS를 계산하기 위해 time 모듈을 임포트합니다.
# [2차]: 1프레임 처리에 몇 밀리초가 걸렸는지 재는 초정밀 스톱워치입니다.
import time

# [1차]: Optional 타입 힌트를 임포트합니다.
# [2차]: 비어 있을 수 있는 변수(예: FeatureMatcher, VideoWriter)를 표시합니다.

# [1차]: VideoCapture, VideoWriter, GUI 창 표시 및 키보드 이벤트 처리를 위해 OpenCV를 임포트합니다.
# [2차]: 영상을 읽고 화면에 띄우고 키보드 입력을 받는 재생기 본체입니다.

# [1차]: 프레임 타입 힌트를 위해 NumPy를 임포트합니다.
# [2차]: 영상 1장이 숫자 배열임을 표시합니다.
import numpy as np

# [1차]: SRP로 분리된 각 하위 파이프라인 모듈을 임포트합니다.
# [2차]: 배경 담당, 감지 담당, 추적 담당, 지문 감식 담당, 화가 팀원들을 한자리에 모읍니다.

# [1차]: 영상 분석 실행 완료 후 핵심 집계 지표를 반환하는 RunSummary 데이터클래스입니다.
# [2차]: 상영이 끝난 뒤 "총 몇 프레임, 평균 FPS, 누적 ID, 타겟 감지 횟수, 스냅샷 수"를 정리한 영수증입니다.
@dataclass
class RunSummary:
    """Summary metrics collected during a video analysis run."""

    frames_processed: int
    average_fps: float
    total_unique_ids: int
    target_detected_frames: int
    snapshots_saved: int


# [1차]: 전체 실시간 영상 분석 루프를 총괄하는 오케스트레이터 클래스 VideoAnalyzerApp입니다.
# [2차]: 모든 팀원(배경·감지·추적·감식·화가)에게 순서대로 일을 지휘하는 감독관입니다.
class VideoAnalyzerApp:
    """Orchestrates VideoCapture, background subtraction, detection, tracking, and feature matching."""

    # [1차]: OpenCV GUI 창에 표시할 고정 타이틀 문자열입니다.
    # [2차]: 모니터에 뜨는 재생 창 맨 위 제목표시줄 글자입니다.
    WINDOW_NAME = "Codyssey A3-02 | Motion Tracker & Re-ID"

    # [1차]: AppConfig 설정을 받아 5개 하위 컴포넌트(배경·감지·추적·시각화·특징점매칭)를 조립합니다.
    # [2차]: 설정표를 보고 각 담당 팀원들에게 알맞은 장비를 나눠주며 출격 준비를 마칩니다.
    def __init__(self, config: AppConfig) -> None:
        self.config = config
        self.bg_subtractor = BackgroundSubtractor(
            method=config.bg_method,
            learning_rate=config.learning_rate,
            history=config.history,
            var_threshold=config.var_threshold,
            morph_kernel_size=config.morph_kernel_size,
        )
        self.detector = MotionDetector(
            min_area=config.min_area,
            max_area=config.max_area,
        )
        self.tracker = Tracker(
            max_disappeared=config.max_disappeared,
            max_distance=config.max_distance,
            max_trajectory_len=config.max_trajectory_len,
            use_velocity_prediction=config.use_velocity_prediction,
        )
        self.visualizer = FrameVisualizer()
        # [1차]: target_image가 설정되어 있고 파일이 실제로 존재할 때만 FeatureMatcher를 활성화합니다.
        # [2차]: 수배 사진이 주어졌을 때만 감식반을 현장에 투입하고, 없으면 추적 업무에만 집중합니다.
        self.feature_matcher: Optional[FeatureMatcher] = None
        if config.target_image:
            target_path = Path(config.target_image)
            if target_path.exists():
                self.feature_matcher = FeatureMatcher(
                    target_image_path=target_path,
                    method=config.feature_method,
                    ratio_thresh=config.ratio_thresh,
                    min_match_count=config.min_match_count,
                )

    # [1차]: 문자열 '0' 같은 숫자는 웹캠 장치 인덱스 int로 변환하고, 파일 경로는 문자열 그대로 유지합니다.
    # [2차]: "0"이라고 적으면 0번 웹캠을 켜고, "vtest.avi"라고 적으면 동영상 파일을 열도록 구분합니다.
    @staticmethod
    def _parse_source(source: str) -> int | str:
        """Convert numeric strings like '0' to webcam index int; keep file paths as str."""
        stripped = source.strip()
        if stripped.isdigit():
            return int(stripped)
        return stripped

    # [1차]: 단일 프레임에 대해 배경차분 → 감지 → 추적 → 특징점매칭 → 시각화 파이프라인을 순차 실행합니다.
    # [2차]: 사진 1장이 들어오면 5명의 팀원이 컨베이어 벨트처럼 차례대로 가공해 완성본을 내놓습니다.
    def process_frame(
        self, frame: np.ndarray, fps: float, paused: bool = False
    ) -> tuple[np.ndarray, Optional[MatchResult]]:
        """Run full CV pipeline on a single frame and return the annotated frame and match result."""
        fg_mask = self.bg_subtractor.apply(frame)
        detections = self.detector.detect(fg_mask)
        tracked = self.tracker.update(detections)

        match_result: Optional[MatchResult] = None
        if self.feature_matcher is not None:
            match_result = self.feature_matcher.match(frame)

        bg_info = (
            f"BG: {self.config.bg_method.upper()} (lr={self.config.learning_rate}) | "
            f"VelPred: {'ON' if self.config.use_velocity_prediction else 'OFF'}"
        )
        rendered = self.visualizer.draw(
            frame=frame,
            tracked_objects=tracked,
            fps=fps,
            match_result=match_result,
            paused=paused,
            bg_info=bg_info,
        )
        return rendered, match_result

    # [1차]: VideoCapture를 열어 프레임 루프를 돌며 FPS 계산, 키보드 제어(q/s/p), 리소스 해제를 수행합니다.
    # [2차]: 재생 버튼을 눌러 영상이 끝날 때까지 실시간 분석을 돌리고, 끝나면 장비를 깔끔하게 정리합니다.
    def run(self) -> RunSummary:
        """Open video source and execute frame-by-frame analysis loop."""
        # [1차]: 파싱된 소스로 OpenCV VideoCapture를 열고 실패 시 RuntimeError를 발생시킵니다.
        # [2차]: 웹캠이나 영상 파일을 열어보고, 안 열리면 즉시 오류를 알립니다.
        capture_source = self._parse_source(self.config.source)
        cap = cv2.VideoCapture(capture_source)
        if not cap.isOpened():
            raise RuntimeError(f"Unable to open video source: {self.config.source}")

        # [1차]: save_video 옵션이 설정된 경우 출력 MP4 VideoWriter를 초기화합니다.
        # [2차]: 녹화 옵션이 켜져 있으면 결과 화면을 저장할 빈 비디오 테이프를 준비합니다.
        writer: Optional[cv2.VideoWriter] = None
        if self.config.save_video:
            out_path = Path(self.config.save_video)
            out_path.parent.mkdir(parents=True, exist_ok=True)
            width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH)) or 640
            height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT)) or 480
            src_fps = cap.get(cv2.CAP_PROP_FPS) or 20.0
            fourcc = cv2.VideoWriter_fourcc(*"mp4v")
            writer = cv2.VideoWriter(str(out_path), fourcc, src_fps, (width, height))

        # [1차]: 통계 카운터, 일시정지 플래그, FPS 측정용 타임스탬프를 초기화합니다.
        # [2차]: 스톱워치를 0으로 맞추고 통계 기록장을 펼칩니다.
        frames_processed = 0
        target_detected_frames = 0
        snapshots_saved = 0
        paused = False
        last_rendered: Optional[np.ndarray] = None
        start_time = time.perf_counter()
        prev_time = start_time

        # [1차]: 예외나 중단 시에도 캡처/라이터/창 리소스가 반드시 해제되도록 try-finally로 감쌉니다.
        # [2차]: 도중에 에러가 나도 카메라와 창을 안전하게 닫고 나가도록 안전벨트를 맵니다.
        try:
            while True:
                # [1차]: 일시정지 상태가 아닐 때만 다음 프레임을 읽고 파이프라인을 실행합니다.
                # [2차]: 재생 중이면 새 장면을 읽어 분석하고, 일시정지 중이면 화면을 멈춰둡니다.
                if not paused:
                    ok, frame = cap.read()
                    if not ok or frame is None:
                        break

                    # [1차]: 직전 프레임과의 경과 시간(0 나눗셈 방지용 1e-6 하한)으로 순간 FPS를 계산합니다.
                    # [2차]: 방금 1장 처리하는 데 걸린 시간으로 초당 몇 장 속도인지 계산합니다.
                    now = time.perf_counter()
                    dt = max(now - prev_time, 1e-6)
                    instant_fps = 1.0 / dt
                    prev_time = now

                    last_rendered, match_result = self.process_frame(
                        frame, fps=instant_fps, paused=False
                    )
                    frames_processed += 1
                    if match_result is not None and match_result.detected:
                        target_detected_frames += 1

                    if writer is not None:
                        writer.write(last_rendered)

                    # [1차]: max_frames 제한에 도달하면 루프를 정상 종료합니다.
                    # [2차]: 미리 정해둔 프레임 수만큼 다 봤으면 재생을 멈춥니다.
                    if (
                        self.config.max_frames is not None
                        and frames_processed >= self.config.max_frames
                    ):
                        break

                # [1차]: GUI 모드일 때 화면을 표시하고 키보드 입력(q/ESC 종료, p 일시정지, s 스냅샷)을 처리합니다.
                # [2차]: 모니터에 결과 화면을 띄우고 사용자의 리모컨 입력(q, p, s)을 실시간으로 받습니다.
                if not self.config.headless and last_rendered is not None:
                    display_frame = (
                        self.visualizer.draw(
                            last_rendered, {}, 0.0, paused=True
                        )
                        if paused
                        else last_rendered
                    )
                    cv2.imshow(self.WINDOW_NAME, display_frame)
                    key = cv2.waitKey(1 if not paused else 30) & 0xFF
                    if key in (ord("q"), 27):
                        break
                    if key == ord("p"):
                        paused = not paused
                    if key == ord("s"):
                        self.visualizer.save_snapshot(
                            last_rendered, self.config.snapshot_dir, frames_processed
                        )
                        snapshots_saved += 1
        finally:
            # [1차]: VideoCapture, VideoWriter, OpenCV GUI 창 리소스를 안전하게 해제합니다.
            # [2차]: 상영이 끝나면 카메라 전원을 끄고 극장 불을 끕니다.
            cap.release()
            if writer is not None:
                writer.release()
            if not self.config.headless:
                cv2.destroyAllWindows()

        # [1차]: 전체 경과 시간 대비 평균 FPS를 계산해 RunSummary로 반환합니다.
        # [2차]: 최종 성적표 영수증을 출력해 호출자에게 돌려줍니다.
        total_elapsed = max(time.perf_counter() - start_time, 1e-6)
        avg_fps = round(frames_processed / total_elapsed, 2)
        return RunSummary(
            frames_processed=frames_processed,
            average_fps=avg_fps,
            total_unique_ids=self.tracker.total_registered,
            target_detected_frames=target_detected_frames,
            snapshots_saved=snapshots_saved,
        )

# [2차]: 윈도우 터미널이 한글을 깨뜨리지 않도록 입구에서 통역기(UTF-8)를 켜주는 방어 코드입니다.
def _ensure_utf8_console() -> None:
    """Protect Windows terminals from cp949 UnicodeEncodeError when printing Korean text."""
    if sys.platform == "win32":
        for stream_name in ("stdout", "stderr"):
            stream = getattr(sys, stream_name, None)
            if stream is not None and hasattr(stream, "reconfigure"):
                try:
                    stream.reconfigure(encoding="utf-8", errors="replace")
                except (OSError, io.UnsupportedOperation):
                    pass
